cmake_minimum_required(VERSION 3.28)
project(nexus_proto LANGUAGES CXX)
include(${CMAKE_CURRENT_LIST_DIR}/../../cmake/NexusCommon.cmake)

# NOT wired into the default build (see the commented-out
# add_subdirectory(libs/proto) at the repo root) — protobuf is fetched
# and built from source here (FetchContent, same pattern as sqlite3's
# amalgamation elsewhere in this repo), and a from-source protobuf build
# is a genuinely slow first build (20-40+ minutes is normal) plus the
# highest-risk dependency this project has taken on for MinGW
# specifically (Abseil version pinning, DLL export issues are the usual
# failure modes). Opt in explicitly by uncommenting that line once
# you're ready to pay that cost, rather than every contributor eating it
# on a build they didn't ask for.
#
# Once opted in: this target (nexus_proto) exposes the generated C++
# bindings for nexus.proto. Nothing in apps/nexus_server or the Python
# client reads/writes this format yet — adopting it means switching
# libs/protocol's line-based text framing over to length-prefixed
# protobuf messages on both ends, which is deliberately a separate,
# follow-up change from just having the schema + codegen available.
include(FetchContent)
FetchContent_Declare(
    protobuf
    GIT_REPOSITORY https://github.com/protocolbuffers/protobuf.git
    GIT_TAG        v27.2
    GIT_SHALLOW    TRUE
)
set(protobuf_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(protobuf_BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(protobuf_MSVC_STATIC_RUNTIME OFF CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(protobuf)

add_library(nexus_proto STATIC)
target_compile_features(nexus_proto PUBLIC cxx_std_23)

protobuf_generate(
    TARGET nexus_proto
    LANGUAGE cpp
    PROTOS ${CMAKE_CURRENT_SOURCE_DIR}/nexus.proto
    IMPORT_DIRS ${CMAKE_CURRENT_SOURCE_DIR}
    PROTOC_OUT_DIR ${CMAKE_CURRENT_BINARY_DIR}/generated
)
target_include_directories(nexus_proto PUBLIC ${CMAKE_CURRENT_BINARY_DIR}/generated)
target_link_libraries(nexus_proto PUBLIC protobuf::libprotobuf)

# Python bindings: not generated by CMake at all — run this by hand
# (needs `pip install protobuf grpcio-tools` first) once you're adopting
# the format on the client side too:
